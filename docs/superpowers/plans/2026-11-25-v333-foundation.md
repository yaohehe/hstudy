Create `AppClock.kt`:

```kotlin
package com.hstudy.app.time

import kotlinx.datetime.Clock
import kotlinx.datetime.Instant
import kotlinx.datetime.TimeZone
import kotlinx.datetime.toLocalDateTime

interface AppClock {
    fun nowMillis(): Long
    fun today(): String = nowMillis().let { millis ->
        Instant.fromEpochMilliseconds(millis)
            .toLocalDateTime(TimeZone.currentSystemDefault())
            .date.toString()
    }
}

object SystemAppClock : AppClock {
    override fun nowMillis(): Long = Clock.System.now().toEpochMilliseconds()
}

class FixedAppClock(var millis: Long) : AppClock {
    override fun nowMillis(): Long = millis
}
```

```kotlin
package com.hstudy.app.data.local

import androidx.room.Transactor

private class IosRoomTransactionRunner(private val database: AppDatabase) : DbTransactionRunner {
```
